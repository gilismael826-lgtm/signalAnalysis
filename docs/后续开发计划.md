# EMG/IMU 动态手势识别系统 - 后续开发计划

## 〇、关键问题：EMG/IMU采样率差异处理

### 0.1 问题描述

本系统中EMG和IMU传感器具有不同的采样率：
- **EMG采样率**: 250 Hz
- **IMU采样率**: 104 Hz

这种差异在特征提取和手势识别时需要特殊处理，否则会导致时间对齐问题和特征计算错误。

### 0.2 解决方案

#### 方案一：时间窗口对齐（推荐）

使用固定时间窗口而非样本数量来对齐数据，EMG和IMU各自维护独立的样本缓冲区：

```python
class RealtimeGestureRecognizer:
    def __init__(self, window_size_ms=600, emg_rate=250, imu_rate=104):
        self.window_size_ms = window_size_ms  # 时间窗口（毫秒）
        
        # 根据采样率计算各自的窗口样本数
        self.emg_window_samples = int(window_size_ms * emg_rate / 1000)  # 150 samples
        self.imu_window_samples = int(window_size_ms * imu_rate / 1000)  # 62 samples
        
        self.emg_window = [[] for _ in range(8)]
        self.imu_window = [[] for _ in range(6)]
    
    def update_emg(self, emg_data):
        """更新EMG窗口"""
        for i in range(8):
            self.emg_window[i].append(emg_data[i])
            if len(self.emg_window[i]) > self.emg_window_samples:
                self.emg_window[i].pop(0)
    
    def update_imu(self, imu_data):
        """更新IMU窗口"""
        for i in range(6):
            self.imu_window[i].append(imu_data[i])
            if len(self.imu_window[i]) > self.imu_window_samples:
                self.imu_window[i].pop(0)
    
    def is_window_ready(self):
        """检查窗口是否已满"""
        return (len(self.emg_window[0]) >= self.emg_window_samples and
                len(self.imu_window[0]) >= self.imu_window_samples)
```

#### 方案二：独立特征提取 + 特征融合

EMG和IMU分别使用各自的采样率进行特征提取，最后合并特征向量：

```python
class FeatureExtractor:
    def __init__(self, emg_sample_rate=250.0, imu_sample_rate=104.0):
        self.emg_sample_rate = emg_sample_rate
        self.imu_sample_rate = imu_sample_rate
    
    def extract_emg_features(self, emg_data):
        """EMG特征提取 - 使用EMG采样率"""
        features = {}
        for i in range(8):
            channel_features = self._extract_channel_features(
                emg_data[i], self.emg_sample_rate, prefix=f'emg_{i}'
            )
            features.update(channel_features)
        return features
    
    def extract_imu_features(self, imu_data):
        """IMU特征提取 - 使用IMU采样率"""
        features = {}
        for i in range(6):
            channel_features = self._extract_channel_features(
                imu_data[i], self.imu_sample_rate, prefix=f'imu_{i}'
            )
            features.update(channel_features)
        return features
    
    def extract_all_features(self, emg_data, imu_data):
        """提取所有特征并融合"""
        features = {}
        features.update(self.extract_emg_features(emg_data))
        features.update(self.extract_imu_features(imu_data))
        return features
    
    def _extract_channel_features(self, data, sample_rate, prefix):
        """提取单通道特征"""
        features = {}
        
        # 时域特征（与采样率无关）
        features[f'{prefix}_mean'] = np.mean(data)
        features[f'{prefix}_std'] = np.std(data)
        features[f'{prefix}_rms'] = np.sqrt(np.mean(np.square(data)))
        features[f'{prefix}_variance'] = np.var(data)
        features[f'{prefix}_peak_to_peak'] = np.max(data) - np.min(data)
        
        # 频域特征（需要采样率）
        n = len(data)
        fft_result = np.fft.fft(data)
        frequencies = np.fft.fftfreq(n, 1/sample_rate)[:n//2]
        power_spectrum = np.abs(fft_result)[:n//2] ** 2 / n
        
        total_power = np.sum(power_spectrum)
        if total_power > 0:
            features[f'{prefix}_median_freq'] = frequencies[
                np.argmax(np.cumsum(power_spectrum) >= total_power/2)
            ]
            features[f'{prefix}_mean_power_freq'] = np.sum(frequencies * power_spectrum) / total_power
            features[f'{prefix}_peak_freq'] = frequencies[np.argmax(power_spectrum)]
        
        return features
```

#### 方案三：重采样对齐（备选）

将IMU数据重采样到与EMG相同的采样率：

```python
from scipy import signal

def resample_imu_to_emg(imu_data, imu_rate=104, emg_rate=250):
    """将IMU数据重采样到EMG采样率"""
    original_samples = len(imu_data)
    target_samples = int(original_samples * emg_rate / imu_rate)
    resampled = signal.resample(imu_data, target_samples)
    return resampled
```

**注意**: 重采样可能引入插值误差，建议优先使用方案一和方案二。

### 0.3 推荐架构

采用**方案一 + 方案二组合**：

1. **数据缓冲层**: 基于时间窗口独立管理EMG和IMU缓冲区
2. **特征提取层**: EMG和IMU分别使用各自采样率计算特征
3. **特征融合层**: 将两种特征合并为一个特征向量
4. **分类层**: 使用融合特征进行手势分类

```python
#!/usr/bin/env python3
"""
特征提取模块
负责从EMG和IMU信号中提取时域、频域和时频域特征
"""

import numpy as np
from scipy import signal
from scipy.stats import entropy
import pywt

class FeatureExtractor:
    """特征提取器"""
    
    def __init__(self, sample_rate=250):
        self.sample_rate = sample_rate
    
    # 时域特征
    def extract_mean(self, data):
        """均值"""
        return np.mean(data)
    
    def extract_variance(self, data):
        """方差"""
        return np.var(data)
    
    def extract_rms(self, data):
        """均方根"""
        return np.sqrt(np.mean(np.square(data)))
    
    def extract_iemg(self, data):
        """积分肌电"""
        return np.sum(np.abs(data))
    
    def extract_zc(self, data, threshold=0):
        """过零率"""
        zc = 0
        for i in range(len(data) - 1):
            if (data[i] - threshold) * (data[i+1] - threshold) < 0:
                zc += 1
        return zc
    
    def extract_ssc(self, data, threshold=0):
        """斜率符号变化率"""
        ssc = 0
        for i in range(1, len(data) - 1):
            diff1 = data[i] - data[i-1]
            diff2 = data[i+1] - data[i]
            if (diff1 * diff2) < 0 and (abs(diff1) > threshold or abs(diff2) > threshold):
                ssc += 1
        return ssc
    
    def extract_wl(self, data):
        """波形长度"""
        return np.sum(np.abs(np.diff(data)))
    
    def extract_wamp(self, data, threshold=0):
        """Willison幅度"""
        wamp = 0
        for i in range(len(data) - 1):
            if abs(data[i] - data[i+1]) > threshold:
                wamp += 1
        return wamp
    
    # 频域特征
    def extract_fft_features(self, data):
        """FFT特征"""
        n = len(data)
        fft_result = np.fft.fft(data)
        frequencies = np.fft.fftfreq(n, 1/self.sample_rate)[:n//2]
        power_spectrum = np.abs(fft_result)[:n//2] ** 2 / n
        
        # 中值频率
        total_power = np.sum(power_spectrum)
        cumulative_power = np.cumsum(power_spectrum)
        mf = frequencies[np.argmax(cumulative_power >= total_power/2)]
        
        # 平均功率频率
        mpf = np.sum(frequencies * power_spectrum) / total_power if total_power > 0 else 0
        
        # 峰值频率
        peak_freq = frequencies[np.argmax(power_spectrum)]
        
        # 频谱能量
        spectral_energy = np.sum(power_spectrum)
        
        # 频谱熵
        psd_norm = power_spectrum / total_power if total_power > 0 else power_spectrum
        spectral_entropy = entropy(psd_norm + 1e-10)
        
        return {
            'median_frequency': mf,
            'mean_power_frequency': mpf,
            'peak_frequency': peak_freq,
            'spectral_energy': spectral_energy,
            'spectral_entropy': spectral_entropy
        }
    
    # 时频域特征
    def extract_wavelet_features(self, data, wavelet='db4', level=3):
        """小波特征"""
        coeffs = pywt.wavedec(data, wavelet, level=level)
        
        features = {}
        for i, coeff in enumerate(coeffs):
            features[f'wavelet_level{i}_mean'] = np.mean(np.abs(coeff))
            features[f'wavelet_level{i}_std'] = np.std(coeff)
            features[f'wavelet_level{i}_energy'] = np.sum(coeff ** 2)
        
        return features
    
    def extract_all_features(self, data):
        """提取所有特征"""
        features = {}
        
        # 时域特征
        features['mean'] = self.extract_mean(data)
        features['variance'] = self.extract_variance(data)
        features['std'] = np.std(data)
        features['rms'] = self.extract_rms(data)
        features['iemg'] = self.extract_iemg(data)
        features['zc'] = self.extract_zc(data)
        features['ssc'] = self.extract_ssc(data)
        features['wl'] = self.extract_wl(data)
        features['wamp'] = self.extract_wamp(data)
        features['peak_to_peak'] = np.max(data) - np.min(data)
        
        # 频域特征
        fft_features = self.extract_fft_features(data)
        features.update(fft_features)
        
        # 时频域特征
        wavelet_features = self.extract_wavelet_features(data)
        features.update(wavelet_features)
        
        return features
```

```python
#!/usr/bin/env python3
"""
特征选择与降维模块
"""

import numpy as np
from sklearn.decomposition import PCA
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis
from sklearn.feature_selection import mutual_info_classif
from sklearn.ensemble import RandomForestClassifier
import matplotlib.pyplot as plt

class FeatureSelector:
    """特征选择器"""
    
    def __init__(self, n_components=10):
        self.n_components = n_components
        self.pca = None
        self.lda = None
    
    def apply_pca(self, X, n_components=None):
        """应用PCA降维"""
        n_components = n_components or self.n_components
        self.pca = PCA(n_components=n_components)
        X_reduced = self.pca.fit_transform(X)
        
        print(f"PCA降维: {X.shape[1]} -> {n_components}")
        print(f"方差解释比例: {self.pca.explained_variance_ratio_}")
        print(f"累计方差解释比例: {np.cumsum(self.pca.explained_variance_ratio_)}")
        
        return X_reduced
    
    def apply_lda(self, X, y, n_components=None):
        """应用LDA降维"""
        n_components = n_components or self.n_components
        self.lda = LinearDiscriminantAnalysis(n_components=min(n_components, len(np.unique(y))-1))
        X_reduced = self.lda.fit_transform(X, y)
        
        print(f"LDA降维: {X.shape[1]} -> {X_reduced.shape[1]}")
        
        return X_reduced
    
    def get_feature_importance(self, X, y):
        """获取特征重要性"""
        # 随机森林特征重要性
        rf = RandomForestClassifier(n_estimators=100, random_state=42)
        rf.fit(X, y)
        rf_importance = rf.feature_importances_
        
        # 互信息
        mi_importance = mutual_info_classif(X, y)
        
        return {
            'random_forest': rf_importance,
            'mutual_information': mi_importance
        }
```

```python
#!/usr/bin/env python3
"""
手势分类模块
负责手势分类器的训练、评估和预测
"""

import numpy as np
from sklearn.svm import SVC
from sklearn.neighbors import KNeighborsClassifier
from sklearn.ensemble import RandomForestClassifier, VotingClassifier
from sklearn.neural_network import MLPClassifier
from sklearn.model_selection import train_test_split, cross_val_score, GridSearchCV
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
from sklearn.metrics import confusion_matrix, classification_report
import joblib
import os

class GestureClassifier:
    """手势分类器"""
    
    def __init__(self, algorithm='svm'):
        """
        初始化分类器
        
        Args:
            algorithm: 分类算法 ('svm', 'knn', 'rf', 'mlp', 'ensemble')
        """
        self.algorithm = algorithm
        self.classifier = None
        self.labels = None
        self.feature_names = None
    
    def _create_classifier(self, params=None):
        """创建分类器"""
        if self.algorithm == 'svm':
            default_params = {'kernel': 'rbf', 'C': 1.0, 'gamma': 'scale', 'probability': True}
            if params:
                default_params.update(params)
            return SVC(**default_params)
        
        elif self.algorithm == 'knn':
            default_params = {'n_neighbors': 5, 'weights': 'distance'}
            if params:
                default_params.update(params)
            return KNeighborsClassifier(**default_params)
        
        elif self.algorithm == 'rf':
            default_params = {'n_estimators': 100, 'max_depth': None, 'random_state': 42}
            if params:
                default_params.update(params)
            return RandomForestClassifier(**default_params)
        
        elif self.algorithm == 'mlp':
            default_params = {
                'hidden_layer_sizes': (100, 50),
                'activation': 'relu',
                'max_iter': 500,
                'random_state': 42
            }
            if params:
                default_params.update(params)
            return MLPClassifier(**default_params)
        
        elif self.algorithm == 'ensemble':
            svm = SVC(kernel='rbf', C=1.0, gamma='scale', probability=True)
            knn = KNeighborsClassifier(n_neighbors=5, weights='distance')
            rf = RandomForestClassifier(n_estimators=100, random_state=42)
            return VotingClassifier(
                estimators=[('svm', svm), ('knn', knn), ('rf', rf)],
                voting='soft'
            )
        
        else:
            raise ValueError(f'Unknown algorithm: {self.algorithm}')
    
    def train(self, X, y, test_size=0.2, params=None):
        """
        训练分类器
        
        Args:
            X: 特征矩阵
            y: 标签向量
            test_size: 测试集比例
            params: 分类器参数
        
        Returns:
            训练结果字典
        """
        self.labels = np.unique(y)
        self.classifier = self._create_classifier(params)
        
        X_train, X_test, y_train, y_test = train_test_split(
            X, y, test_size=test_size, random_state=42, stratify=y
        )
        
        self.classifier.fit(X_train, y_train)
        
        y_pred = self.classifier.predict(X_test)
        
        results = {
            'accuracy': accuracy_score(y_test, y_pred),
            'precision': precision_score(y_test, y_pred, average='weighted'),
            'recall': recall_score(y_test, y_pred, average='weighted'),
            'f1': f1_score(y_test, y_pred, average='weighted'),
            'confusion_matrix': confusion_matrix(y_test, y_pred),
            'classification_report': classification_report(y_test, y_pred)
        }
        
        cv_scores = cross_val_score(self.classifier, X, y, cv=5)
        results['cv_scores'] = cv_scores
        results['cv_mean'] = np.mean(cv_scores)
        results['cv_std'] = np.std(cv_scores)
        
        return results
    
    def predict(self, X):
        """预测手势"""
        if self.classifier is None:
            raise ValueError('Classifier not trained')
        return self.classifier.predict(X)
    
    def predict_proba(self, X):
        """预测手势概率"""
        if self.classifier is None:
            raise ValueError('Classifier not trained')
        return self.classifier.predict_proba(X)
    
    def save_model(self, filepath):
        """保存模型"""
        if self.classifier is None:
            raise ValueError('Classifier not trained')
        
        model_data = {
            'classifier': self.classifier,
            'labels': self.labels,
            'algorithm': self.algorithm,
            'feature_names': self.feature_names
        }
        joblib.dump(model_data, filepath)
        print(f"模型已保存到: {filepath}")
    
    def load_model(self, filepath):
        """加载模型"""
        model_data = joblib.load(filepath)
        self.classifier = model_data['classifier']
        self.labels = model_data['labels']
        self.algorithm = model_data['algorithm']
        self.feature_names = model_data.get('feature_names')
        print(f"模型已加载: {filepath}")
```

**代码集成示例**:
```python
# 在serial_comm.py中添加

from src.feature_extractor import FeatureExtractor
from src.gesture_classifier import GestureClassifier

class RealtimeGestureRecognizer:
    """实时手势识别器"""
    
    def __init__(self, model_path, window_size=600, sample_rate=250):
        self.feature_extractor = FeatureExtractor(sample_rate=sample_rate)
        self.classifier = GestureClassifier()
        self.classifier.load_model(model_path)
        
        self.window_size = window_size  # 窗口大小（ms）
        self.sample_rate = sample_rate
        self.window_samples = int(window_size * sample_rate / 1000)
        
        self.emg_window = [[] for _ in range(8)]
        self.imu_window = [[] for _ in range(6)]
        
        self.prediction_history = []
        self.history_size = 5
    
    def update(self, emg_data, imu_data):
        """更新窗口数据并预测"""
        # 更新EMG窗口
        for i in range(8):
            self.emg_window[i].append(emg_data[i])
            if len(self.emg_window[i]) > self.window_samples:
                self.emg_window[i].pop(0)
        
        # 更新IMU窗口
        for i in range(6):
            self.imu_window[i].append(imu_data[i])
            if len(self.imu_window[i]) > self.window_samples:
                self.imu_window[i].pop(0)
        
        # 检查窗口是否已满
        if len(self.emg_window[0]) < self.window_samples:
            return None, 0
        
        # 提取特征
        features = {}
        for i in range(8):
            channel_features = self.feature_extractor.extract_all_features(
                np.array(self.emg_window[i])
            )
            for key, value in channel_features.items():
                features[f'emg_{i}_{key}'] = value
        
        for i in range(6):
            channel_features = self.feature_extractor.extract_all_features(
                np.array(self.imu_window[i])
            )
            for key, value in channel_features.items():
                features[f'imu_{i}_{key}'] = value
        
        # 预测
        X = np.array([list(features.values())])
        prediction = self.classifier.predict(X)[0]
        probabilities = self.classifier.predict_proba(X)[0]
        confidence = np.max(probabilities)
        
        # 结果平滑
        self.prediction_history.append(prediction)
        if len(self.prediction_history) > self.history_size:
            self.prediction_history.pop(0)
        
        # 多数投票
        from collections import Counter
        vote_result = Counter(self.prediction_history).most_common(1)[0]
        final_prediction = vote_result[0]
        final_confidence = confidence
        
        return final_prediction, final_confidence
```
